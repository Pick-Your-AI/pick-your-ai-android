def git(kind,data):
    return hashlib.sha1(kind.encode()+b' '+str(len(data)).encode()+b'\0'+data).hexdigest()
def metadata(data):
    return {'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest(),'git_blob':git('blob',data),'mode':'100644'}
def write(path,data):
    p=out/path;p.parent.mkdir(parents=True,exist_ok=True);p.write_bytes(data)
def tree_hash(entries):
    children={}
    for path,obj in entries.items():
        parts=path.split('/');node=children
        for part in parts[:-1]:node=node.setdefault(part,{})
        node[parts[-1]]=obj
    def walk(node):
        records=[]
        for name,value in node.items():
            directory='sha' not in value
            mode='40000' if directory else value['mode']
            sha=walk(value) if directory else value['sha']
            records.append((name+'/' if directory else name,mode.encode()+b' '+name.encode()+b'\0'+bytes.fromhex(sha)))
        return git('tree',b''.join(data for _,data in sorted(records)))
    return walk(children)

# First join the separately reviewed dataset/framework and fine-model insertion families on compute.
proofs=[]
def compose(base,first,second,path,scope):
    lines=base.splitlines(keepends=True);families=[];insertions=[]
    for family_id,source in enumerate((first,second)):
        changed=source.splitlines(keepends=True);family=[]
        for tag,i,j,k,l in difflib.SequenceMatcher(a=lines,b=changed,autojunk=False).get_opcodes():
            if tag=='equal':continue
            assert tag=='insert' and i==j,(path,scope,tag)
            part=''.join(changed[k:l]);family.append(part);insertions.append((i,family_id,part))
        restored=source
        for part in family:
            assert restored.count(part)==1,(path,'ambiguous insertion')
            restored=restored.replace(part,'',1)
        assert restored==base
        families.append(family)
    result='';previous=0
    for position,family_id,part in sorted(insertions):
        result+=''.join(lines[previous:position])+part;previous=position
    result+=''.join(lines[previous:])
    for family,expected in zip(families,(second,first)):
        restored=result
        for part in family:
            assert restored.count(part)==1
            restored=restored.replace(part,'',1)
        assert restored==expected,(path,'whole opposite source inverse')
    restored=result
    for family in families:
        for part in family:restored=restored.replace(part,'',1)
    assert restored==base
    proofs.append({'scope':scope,'path':path,'base_blob':git('blob',base.encode()),
        'first_blob':git('blob',first.encode()),'second_blob':git('blob',second.encode()),
        'combined_blob':git('blob',result.encode()),'insertion_families':[len(f) for f in families],
        'both_cross_inverses_and_base_inverse_exact':True})
    return result
